function getDate(v){
  if(!v) return null;
  return v.toDate ? v.toDate() : new Date(v);
}

function formatYMD(v){
  const d = getDate(v);
  if(!d || isNaN(d)) return "";
  const y = d.getFullYear();
  const m = String(d.getMonth()+1).padStart(2,"0");
  const day = String(d.getDate()).padStart(2,"0");
  return `${y}-${m}-${day}`;
}

function num(v){
  return Number(v || 0) || 0;
}

function money(v){
  return num(v).toFixed(2);
}

function safeText(v){
  return String(v ?? "").trim();
}

function toMinutes(hhmm){
  if(!hhmm || !hhmm.includes(":")) return 0;
  const [h,m] = hhmm.split(":").map(Number);
  return (h * 60) + m;
}

function minutesToHours(mins){
  return num(mins) / 60;
}

function timeStringFromTS(ts){

  if(!ts) return "";

  let d = null;

  if(ts.toDate){
    d = ts.toDate();
  }else{
    d = new Date(ts);
  }

  if(!d || isNaN(d)){
    return "";
  }

  return d.toTimeString().slice(0,5);

}

function calcPayrollMetrics(record, course){
  const empty={actualMinutes:0,grossMinutes:0,delayMinutes:0,deductionMinutes:0,workedMinutes:0};
  const parseDate=v=>{const d=v?.toDate?v.toDate():new Date(v);return Number.isFinite(d.getTime())?d:null;};
  const entry=parseDate(record?.startTS),exit=parseDate(record?.endTS);
  if(!entry||!exit)return empty;
  let inMs=entry.getTime(),outMs=exit.getTime();
  if(outMs<=inMs && inMs-outMs<86400000)outMs+=86400000;
  if(outMs<=inMs)return empty;
  const actualMinutes=Math.min(1440,(outMs-inMs)/60000);
  const st=safeText(course?.courseStartTime),en=safeText(course?.courseEndTime);
  if(!/^\d{1,2}:\d{2}$/.test(st)||!/^\d{1,2}:\d{2}$/.test(en)){
    // Without official hours, do not invent a cap or lateness deduction.
    return {...empty,actualMinutes:Math.round(actualMinutes),grossMinutes:Math.round(actualMinutes),workedMinutes:Math.round(actualMinutes)};
  }
  const day=/^\d{4}-\d{2}-\d{2}$/.test(String(record.Date||""))?record.Date:null;
  const base=day?new Date(day+"T00:00:00"):new Date(entry.getFullYear(),entry.getMonth(),entry.getDate());
  const [sh,sm]=st.split(":").map(Number),[eh,em]=en.split(":").map(Number);
  const start=new Date(base);start.setHours(sh,sm,0,0);
  const end=new Date(base);end.setHours(eh,em,0,0);
  if(end<=start)end.setDate(end.getDate()+1);
  if(!day && inMs<start.getTime() && start.getTime()-inMs>43200000){start.setDate(start.getDate()-1);end.setDate(end.getDate()-1);}
  const officialStart=start.getTime(),officialEnd=end.getTime();
  const duration=(officialEnd-officialStart)/60000;
  // Maximum eligible time is the scheduled interval up to the trainer's checkout.
  const grossMinutes=Math.max(0,Math.min(duration,(Math.min(outMs,officialEnd)-officialStart)/60000));
  const delayMinutes=Math.max(0,(inMs-officialStart)/60000);
  // 0-15 min inclusive: grace. Over 15 min: charge every late minute.
  const deductionMinutes=delayMinutes>15?Math.min(grossMinutes,delayMinutes):0;
  const workedMinutes=Math.max(0,grossMinutes-deductionMinutes);
  return {actualMinutes:Math.round(actualMinutes),grossMinutes:Math.round(grossMinutes),delayMinutes:Math.round(delayMinutes),deductionMinutes:Math.round(deductionMinutes),workedMinutes:Math.round(workedMinutes)};
}

db.collection("Courses").onSnapshot(s=>{
  ALL_COURSES = [];

  const sets = {
    Trainer:new Set(),
    Course:new Set(),
    Period:new Set(),
    Section:new Set(),
    Branch:new Set(),
    Start:new Set()
  };

  s.forEach(doc=>{
    const c = doc.data() || {};
    if(c.status === "closed") return;

    const row = { id:doc.id, ...c };
    ALL_COURSES.push(row);

    if(c.Trainer) sets.Trainer.add(c.Trainer);
    if(c.TrainingCourse) sets.Course.add(c.TrainingCourse);
    if(c.Period) sets.Period.add(c.Period);
    if(c.Section) sets.Section.add(c.Section);
    if(c.Branch) sets.Branch.add(c.Branch);

    const sd = formatYMD(c.StartingDate);
    if(sd) sets.Start.add(sd);
  });

  fillSelect("fTrainer", sets.Trainer, "المدرب");
  fillSelect("fCourse", sets.Course, "الدورة");
  fillSelect("fPeriod", sets.Period, "الفترة");
  fillSelect("fSection", sets.Section, "القسم");
  fillSelect("fBranch", sets.Branch, "الفرع");
  fillSelect("fStart", sets.Start, "تاريخ البداية");

  applyFilters();
});

function fillSelect(id, set, title){
  const el = document.getElementById(id);
  const oldVal = el.value;
  el.innerHTML = `<option value="">${title}</option>`;
  [...set].sort((a,b)=>String(a).localeCompare(String(b),"ar")).forEach(v=>{
    el.innerHTML += `<option value="${String(v).replace(/"/g,'&quot;')}">${v}</option>`;
  });
  if([...el.options].some(o=>o.value === oldVal)) el.value = oldVal;
}

function applyFilters(){
  const t = document.getElementById("fTrainer").value;

  const s = document.getElementById("fSection").value;
  const b = document.getElementById("fBranch").value;

  if(!t && !c && !p && !s && !b && !d){
    FILTERED_COURSES = [];
    document.getElementById("coursesMulti").innerHTML = "";
    document.getElementById("note").style.display = "block";
    document.getElementById("tools").style.display = "none";
    document.getElementById("summaryPreview").style.display = "none";
    return;
  }

  FILTERED_COURSES = ALL_COURSES.filter(x=>{
    if(t && x.Trainer !== t) return false;
    if(c && x.TrainingCourse !== c) return false;

    if(s && x.Section !== s) return false;

    if(d && formatYMD(x.StartingDate) !== d) return false;
    return true;
  });

  renderCourses(FILTERED_COURSES);
}

async function saveHourPrice(courseId){
  const input = document.getElementById("price_" + courseId);
  if(!input) return;

  const newPrice = num(input.value);

  try{
    await db.collection("Courses").doc(courseId).set({
      HourPrice:newPrice,
      updatedAt:firebase.firestore.FieldValue.serverTimestamp(),
      updatedBy:localStorage.getItem("username") || "system"
    }, {merge:true});

    const course = ALL_COURSES.find(x=>x.id === courseId);
    if(course){
      course.HourPrice = newPrice;
      course.hourPrice = newPrice;
    }

    const currentPriceEl = document.getElementById("currentPrice_" + courseId);
    if(currentPriceEl) currentPriceEl.textContent = `${money(newPrice)} ريال`;

    alert("تم حفظ سعر الساعة بنجاح");
  }catch(err){
    console.error(err);
    alert("حدث خطأ أثناء حفظ سعر الساعة");
  }
}
